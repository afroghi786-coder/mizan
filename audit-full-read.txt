317: async function cachedRead<T>(key: string, fetcher: () => Promise<T>): Promise<T> {
318:   if (MODE === 'online') {
319:     try {
320:       const fresh = await fetcher();
321:       await setLocal(key, fresh);
322:       return fresh;
323:     } catch {
324:       const cached = await getLocal<T>(key);
325:       if (cached) return cached;
326:       throw new Error('آفلاین — داده‌ای موجود نیست');
327:     }
328:   }
329:   const cached = await getLocal<T>(key);
330:   if (cached) return cached;
331:   try { return await fetcher(); } catch { throw new Error('داده‌ای موجود نیست'); }
332: }
