159: export const TAX_BRACKETS = [
160:   { min: 0, max: 5000000, rate: 0 },
161:   { min: 5000000, max: 10000000, rate: 10 },

168: export function calcTax(monthlySalary: number): number {
169:   const annual = monthlySalary * 12;
170:   let tax = 0;
171:   TAX_BRACKETS.forEach(b => {
172:     if (annual > b.min) {
173:       const taxable = Math.min(annual, b.max) - b.min;
174:       if (taxable > 0) tax += taxable * (b.rate / 100);
175:     }
176:   });
177:   return Math.round(tax / 12);
178: }

202:     insurance_rate: Math.max(0, Number(settings?.insurance_rate ?? 7)),

204:     auto_tax: settings?.auto_tax !== false,
205:     loan_deduction: settings?.loan_deduction !== false,
206:     overtime_rate_multiplier: Math.max(0, Number(settings?.overtime_rate_multiplier ?? 1.4)),

218:     if (l.employee_code !== emp.code) return false;
219:     if (l.status && l.status !== 'approved') return false;
220:     const from = String(l.from_date || '');
221:     const to = String(l.to_date || '');
222:     return from.startsWith(month) || to.startsWith(month);
223:   });

226:   const unpaidLeaveDays = myLeaves.filter((l: any) => l.leave_type === 'unpaid').reduce((s: number, l: any) => s + (Number(l.days) || 0), 0);
227:   const totalLeave = leaveFromAtt + annualLeaveDays + sickLeaveDays + unpaidLeaveDays;
228: 
229:   const type = emp.contract_type || 'ماهانه';

255:

262:   const paidLeaveDeduction = cfg.deduct_paid_leave ? Math.round(annualLeaveDays * dailyRate * cfg.paid_leave_rate) : 0;
263:   const sickLeaveDeduction = cfg.deduct_sick_leave ? Math.round(sickLeaveDays * dailyRate * cfg.sick_leave_rate) : 0;
264: 
265:   const totalOtherDeductions = absentDeduction + unpaidLeaveDeduction + paidLeaveDeduction + sickLeaveDeduction;
266:   const totalDeductions = insurance + tax + loanDeduction + totalOtherDeductions;
267:   const net = Math.max(0, gross - totalDeductions);
268: 
269:   const mySalaries = (salaries || []).filter((x: any) => {
270:     if (x.employee_code !== emp.code) return false;
271:     const m = String(x.month || '');
272:     const d = String(x.payment_date || '');
273:     return m.startsWith(month) || d.startsWith(month);
274:   });
275:   const alreadyPaid = mySalaries.reduce((acc: number, x: any) => acc + (Number(x.payment) || 0), 0);
276:   const remaining = Math.max(0, net - alreadyPaid);
277: 
278:   return {
279:     attendance_days: present, absent_days: absent,
280:     leave_days: totalLeave,
281:     paid_leave_days: annualLeaveDays + sickLeaveDays,
282:     unpaid_leave_days: unpaidLeaveDays,
283:     annual_leave_days: annualLeaveDays,
284:     sick_leave_days: sickLeaveDays,
285:     worked_hours: totalHours, overtime_hours: totalOT,
286:     base_salary: base,
287:     housing_allowance: housing, food_allowance: food,
288:     transport_allowance: transport, other_allowance: other, overtime: ot,
289:     gross_salary: gross,
290:     employee_insurance: insurance, employer_insurance: employerInsurance, tax,
291:     loan: loanDeduction,
292:     deduction: absentDeduction,
293:     absence_deduction: absentDeduction,
294:     unpaid_leave_deduction: unpaidLeaveDeduction,
295:     paid_leave_deduction: paidLeaveDeduction,
296:     sick_leave_deduction: sickLeaveDeduction,
297:     total_other_deductions: totalOtherDeductions,
